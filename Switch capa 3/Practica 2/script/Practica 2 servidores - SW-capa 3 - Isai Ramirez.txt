!------------------- RT-01 DERECHO --------------------------

hostname RT-01-Derecho

no shutdown
exit

exit

interface g0/0.3

ip address 172.16.3.254 255.255.255.0
exit


interface s0/0/0

no shutdown

!-----------------------------------------------------

no shutdown
ip address 172.16.4.254 255.255.255.0
exit

exit
