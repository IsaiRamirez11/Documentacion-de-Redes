!------------------- RT-01 IZQUIERDO --------------------------

ena
conf term
hostname RT-01

interface g0/0
no shutdown
exit

interface g0/0.6
encapsulation dot1Q 6
ip address 192.168.6.254 255.255.255.0
exit

interface g0/0.7
encapsulation dot1Q 7
ip address 192.168.7.254 255.255.255.0
exit


interface s0/0/0
ip address 10.2.2.1 255.255.255.252
no shutdown



!---------- DHCP DE LA VLAN 6 -------------------------

ip dhcp excluded-address 192.168.6.1 192.168.6.10
ip dhcp excluded-address 192.168.6.252 192.168.6.254

ip dhcp pool VLAN6
network 192.168.6.0 255.255.255.0
default-router 192.168.6.254
dns-server 192.168.6.252
domain-name isai.com
exit

!---------- DHCP DE LA VLAN 7 -------------------------


ip dhcp excluded-address 192.168.7.1 192.168.7.10
ip dhcp excluded-address 192.168.7.252 192.168.7.254

ip dhcp pool VLAN7
network 192.168.7.0 255.255.255.0
default-router 192.168.7.254
dns-server 192.168.7.252
domain-name isai.com
exit

! ------------------------ RIP -----------------------------------

router rip
version 2
no auto-summary
network 192.168.6.0
network 192.168.7.0
network 10.0.0.0
exit

end
copy run start
